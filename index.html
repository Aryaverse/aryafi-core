module AryaFi.Finance.Cash where import AryaFi.Common.Identifiers (AccountId, CorrelationId) import AryaFi.Common.Types (Amount, Currency) import AryaFi.Common.Errors (require, requirePositive, requireNonEmpty) template CashPosition with bank : Party owner : Party accountId : AccountId currency : Currency amount : Amount isLocked : Bool lockReference : Optional Text observers : [Party] where signatory bank, owner observer observers ensure amount > 0.0 key (bank, accountId, currency) : (Party, AccountId, Currency) maintainer key._1 choice SplitCash : (ContractId CashPosition, ContractId CashPosition) with splitAmount : Amount controller owner do require (not isLocked) "Cannot split a locked cash position" requirePositive splitAmount "splitAmount" require (splitAmount < amount) ("Split amount must be strictly less than total " <> show amount) c1 <- create this with amount = splitAmount c2 <- create this with amount = amount - splitAmount pure (c1, c2) choice MergeCash : ContractId CashPosition with otherCid : ContractId CashPosition controller owner do require (not isLocked) "Cannot merge: this position is locked" other <- fetch otherCid require (other.bank == bank) "Bank mismatch" require (other.owner == owner) "Owner mismatch" require (other.currency == currency) "Currency mismatch" archive otherCid create this with amount = amount + other.amount
